cmake_minimum_required(VERSION 3.28)
project(Kodex VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
list(APPEND CMAKE_PREFIX_PATH "${CMAKE_SOURCE_DIR}/.deps/usr")
find_package(ECM 6.30 REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs6)
include(CTest)
find_package(Qt6 6.11 REQUIRED COMPONENTS Core Gui Widgets Qml Quick QuickControls2 Network Test)
find_package(KF6 6.30 REQUIRED COMPONENTS CoreAddons Config I18n Kirigami Wallet SyntaxHighlighting)
qt_standard_project_setup(REQUIRES 6.11)
add_library(kodex_core STATIC
    src/settings.cpp src/rpcclient.cpp src/listmodel.cpp src/gitservice.cpp
    src/maintenance.cpp src/appcontroller.cpp src/richtext.cpp)
target_include_directories(kodex_core PUBLIC src)
target_link_libraries(kodex_core PUBLIC Qt6::Core Qt6::Gui Qt6::Qml Qt6::Network
    KF6::ConfigCore KF6::I18n KF6::Wallet KF6::SyntaxHighlighting)
target_compile_options(kodex_core PRIVATE -Wall -Wextra -Wpedantic)
qt_add_executable(kodex src/main.cpp)
qt_add_qml_module(kodex URI org.kde.kodex VERSION 1.0
    QML_FILES qml/Main.qml qml/SettingsPage.qml qml/RequestCard.qml qml/GitPanel.qml qml/InlineSelector.qml)
target_link_libraries(kodex PRIVATE kodex_core Qt6::Widgets Qt6::Quick Qt6::QuickControls2 KF6::Kirigami KF6::I18nQml KF6::CoreAddons)
install(TARGETS kodex ${KDE_INSTALL_TARGETS_DEFAULT_ARGS})
install(FILES data/org.kde.kodex.desktop DESTINATION ${KDE_INSTALL_APPDIR})
install(FILES data/org.kde.kodex.svg DESTINATION ${KDE_INSTALL_ICONDIR}/hicolor/scalable/apps)
find_program(MSGFMT_EXECUTABLE msgfmt REQUIRED)
add_custom_command(OUTPUT ${CMAKE_BINARY_DIR}/locale/ru/LC_MESSAGES/kodex.mo
    COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_BINARY_DIR}/locale/ru/LC_MESSAGES
    COMMAND ${MSGFMT_EXECUTABLE} ${CMAKE_SOURCE_DIR}/po/ru.po -o ${CMAKE_BINARY_DIR}/locale/ru/LC_MESSAGES/kodex.mo
    DEPENDS po/ru.po)
add_custom_target(translations ALL DEPENDS ${CMAKE_BINARY_DIR}/locale/ru/LC_MESSAGES/kodex.mo)
install(FILES ${CMAKE_BINARY_DIR}/locale/ru/LC_MESSAGES/kodex.mo DESTINATION ${KDE_INSTALL_LOCALEDIR}/ru/LC_MESSAGES)
if(BUILD_TESTING)
    qt_add_executable(kodex_tests tests/test_core.cpp)
    target_link_libraries(kodex_tests PRIVATE kodex_core Qt6::Test)
    target_compile_definitions(kodex_tests PRIVATE SOURCE_DIR="${CMAKE_SOURCE_DIR}")
    add_test(NAME core COMMAND kodex_tests)
    set_tests_properties(core PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    qt_add_executable(kodex_live_tests tests/test_live.cpp)
    target_link_libraries(kodex_live_tests PRIVATE kodex_core Qt6::Test)
    qt_add_executable(kodex_interactive_tests tests/test_interactive.cpp)
    target_compile_definitions(kodex_interactive_tests PRIVATE SOURCE_DIR="${CMAKE_SOURCE_DIR}")
    target_link_libraries(kodex_interactive_tests PRIVATE kodex_core Qt6::Test Qt6::Widgets Qt6::Quick Qt6::QuickControls2 KF6::Kirigami KF6::I18nQml KF6::CoreAddons)
endif()
